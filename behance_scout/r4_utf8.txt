       setExcludedItems(prev => ({
         ...prev,
         [categoryId]: prev[categoryId].filter(id => id !== itemId)

      if (catSelections.includes(itemId)) {
        const newWeights = { ...itemWeights };
        delete newWeights[`${categoryId}-${itemId}`];
        setItemWeights(newWeights);
        return { ...prev, [categoryId]: catSelections.filter(id => id !== itemId) };
      } else {
        return { ...prev, [categoryId]: [...catSelections, itemId] };
      }
    });
  };

  const handleExclude = (categoryId, itemId) => {
    if (selectedItems[categoryId]?.includes(itemId)) {
      setSelectedItems(prev => ({
        ...prev,
        [categoryId]: prev[categoryId].filter(id => id !== itemId)
      }));
      const newWeights = { ...itemWeights };
      delete newWeights[`${categoryId}-${itemId}`];
      setItemWeights(newWeights);
    }
    setExcludedItems(prev => {
      const catExclusions = prev[categoryId] || [];
      if (catExclusions.includes(itemId)) {

        return { ...prev, [categoryId]: [...catExclusions, itemId] };
      }
    });

           itemStrs.forEach(itemStr => {
             const [idStr, weightStr] = itemStr.split('::');

                   newWeights[`${catId}-${id}`] = parseInt(weightStr.trim());
                 }
               }
             }
           });
        }
      }
    });

    setExcludedItems(newExclusions);
    setItemWeights(newWeights);

                          </div>

                    </div>

                    <div className="grid grid-cols-2 gap-x-12 gap-y-10 mb-12">
